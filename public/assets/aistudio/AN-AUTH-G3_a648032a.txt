# AN-AUTH-G3 — Login UI dual path (A.N when flag on) (LEAN)
# When GET /api/an/auth/login-target returns anLoginFlag=true, Login can use POST /api/an/auth/login.
# Default when flag off: unchanged Firebase verify-credentials only.
# Does NOT delete Firebase path. Does NOT set cutoverApproved.

## HARD RULES
1. If login-target fetch fails or anLoginFlag is false → behavior identical to pre-G3 (Firebase only).
2. When anLoginFlag true, offer mode toggle: «A.N» | «Firebase» (labels AR/EN). Default mode = **an**.
3. Never remove `/api/auth/verify-credentials` branch.
4. Do not store password. Access/refresh tokens may be stored in sessionStorage keys `an_access_token` / `an_refresh_token` only.
5. Stop at first failure. tsc wait >2 min → once → build → 9–12. One RESULTS REPORT only.

## PRECHECK
- G2: POST `/api/an/auth/login` exists.
- Login.tsx uses verify-credentials.

### 3b — useEffect on mount

```typescript
    let cancelled = false;
    (async () => {
      try {
        const res = await fetch('/api/an/auth/login-target');
        const data = await res.json().catch(() => ({}));
        if (!cancelled && data?.anLoginFlag === true) {
          setAnLoginAvailable(true);
          setLoginMode('an');
        }
      } catch {
        /* keep firebase-only */
      }
    })();
    return () => {
      cancelled = true;
    };
  }, []);
```

```typescript
import fs from 'fs';
import path from 'path';

let passed = 0;
let failed = 0;
function check(name: string, cond: boolean, detail = '') {
  if (cond) {
    console.log(`[PASS] ${name}`);
    passed++;
  } else {
    console.log(`[FAIL] ${name}${detail ? ' — ' + detail : ''}`);
    failed++;
  }
}

const root = process.cwd();
const login = fs.readFileSync(path.join(root, 'src/pages/Login.tsx'), 'utf8');
const server = fs.readFileSync(path.join(root, 'server.ts'), 'utf8');

check('S1 login-target fetch in Login', login.includes('/api/an/auth/login-target'));

check('S3 verify-credentials retained', login.includes('/api/auth/verify-credentials'));
check('S4 loginMode an|firebase', login.includes("'an'") && login.includes('firebase'));
check('S5 anLoginFlag check', login.includes('anLoginFlag'));
check('S6 sessionStorage an_access_token', login.includes('an_access_token'));
check('S7 server an login route', server.includes('/api/an/auth/login'));
check('S8 server verify-credentials', server.includes('/api/auth/verify-credentials'));
check('S9 MERGE G3', statusDoc.includes('G3'));
check('S10 no cutoverApproved true in Login', !login.includes('cutoverApproved: true'));
check('S11 onLoginSuccess still used', login.includes('onLoginSuccess'));
check('S12 tenantId for an path', login.includes('tenantId'));
```
