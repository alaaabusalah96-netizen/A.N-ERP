```typescript
export interface StockBalanceRow {
  warehouseId: string;
  itemId: string;
  currentQty: number;
  reservedQty?: number;
  availableQty?: number;
  averageCost?: number;
  totalValue?: number;
}

export async function listAnStockBalances(
  adapter: StoragePort,
  companyId: string,
  opts?: { warehouseId?: string; itemId?: string },
  resolution?: CompanyAdapterResolution
): Promise<InventoryListResult<StockBalanceRow>> {
  try {
    const res = await adapter.listWarehouseStock(
      companyId,
      opts?.warehouseId,
      opts?.itemId
    );
    if (!res.success) {
      return {
        ok: false,
        companyId,
        count: 0,
        items: [],
        error: res.error || 'listWarehouseStock failed',
      };
    }
    const items: StockBalanceRow[] = (res.data || []).map((s: any) => ({
      warehouseId: s.warehouseId,
      itemId: s.itemId,
      currentQty: Number(s.currentQty ?? 0),
      reservedQty: s.reservedQty != null ? Number(s.reservedQty) : undefined,
      availableQty: s.availableQty != null ? Number(s.availableQty) : undefined,
      averageCost: s.averageCost != null ? Number(s.averageCost) : undefined,
      totalValue: s.totalValue != null ? Number(s.totalValue) : undefined,
    }));
    return {
      ok: true,
      companyId,
      count: items.length,
      items,
      adapterResolution: resolution
        ? {
            source: resolution.source,
            adapterType: resolution.adapterType,
            isolationLevel: resolution.isolationLevel,
            connectionRef: resolution.connectionRef,
          }
        : undefined,
    };
  } catch (e: any) {
    return {
      ok: false,
      companyId,
      count: 0,
      items: [],
      error: (e?.message || String(e)).slice(0, 200),
    };
  }
}
```

## STEP 3b — `server.ts`
```javascript
    app.get("/api/an/inventory/stock", async (req: any, res: any) => {
      try {
        const authResult = await authenticateSessionOrBearer(req);
        if (!authResult.authenticated) {
          return res.status(authResult.statusCode || 401).json({ ok: false, message: authResult.message });
        }
        const companyId = String(req.query.companyId || "").trim();
        if (!companyId) return res.status(400).json({ ok: false, error: "companyId required" });
        const warehouseId = req.query.warehouseId ? String(req.query.warehouseId).trim() : undefined;
        const itemId = req.query.itemId ? String(req.query.itemId).trim() : undefined;
        const { loadCompanyLikeFromFirestore, resolveCompanyStorageAdapter } = await import("./src/lib/anResolveCompanyAdapter");
        const { listAnStockBalances } = await import("./src/lib/anInventoryLists");
        const companyLike = await loadCompanyLikeFromFirestore(
          typeof adminDb !== "undefined" ? adminDb : null,
          companyId
        );
        let resolved;
        try {
          resolved = resolveCompanyStorageAdapter(anStorageRouter, companyId, companyLike, null);
        } catch (e: any) {
          return res.status(400).json({ ok: false, code: "adapter_resolve_failed", message: e?.message || String(e) });
        }
        const result = await listAnStockBalances(
          resolved.adapter,
          companyId,
          { warehouseId, itemId },
          resolved
        );
        res.status(result.ok ? 200 : 500).json(result);
      } catch (err: any) {
        res.status(500).json({ ok: false, error: err?.message || String(err) });
      }
    });
```
