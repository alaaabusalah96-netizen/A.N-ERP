# AN-MIGRATE-WRITE-F2 — Inventory warehouses write migrate (gated) (LEAN)
# Full merge continues after F1 dry-run.
# Writes warehouses into A.N primary adapter ONLY when AN_MIGRATE_WRITES=1.
# Does NOT switch Inventory.tsx. Does NOT set cutoverApproved. Does NOT migrate stock balances.

## HARD RULES
1. If `process.env.AN_MIGRATE_WRITES` is not `1` or `true`: respond 403 with writesPerformed=false and do nothing.
2. Prefer postgresql primary adapter; if process plane is not postgresql+env ready, refuse write.
3. Idempotent: skip if warehouse id already exists on A.N (listWarehouses then save only missing).
4. Inventory.tsx stays on CustodyDB. Login unchanged.
5. Stop at first failure. tsc wait >2 min → once → build → 9–12. One RESULTS REPORT only.

## PRECHECK
- F1 dry-run route exists.
- StoragePort `saveWarehouse(tenantId, warehouse, idempotencyKey)`.

## STEP 3 — Create `src/lib/anMigrateInventoryWrite.ts`

```typescript
/**
 * AN-MIGRATE-WRITE-F2 — gated warehouse upsert into A.N storage.
 */
import type { StorageRouter } from '../an-core/storage/StorageRouter';
import type { Warehouse } from '../an-core/storage/StoragePort';
import { probeDefaultProcessPlane } from './anDataPlane';

export interface KemitWarehouseRaw {
  id?: string;
  company_id?: string;
  code?: string;
  name?: string;
  nameAr?: string;
  name_ar?: string;
  nameEn?: string;
  isActive?: boolean;
  is_active?: boolean;
  location?: string;
  [key: string]: unknown;
}

export function mapKemitWarehouseToAn(
  raw: KemitWarehouseRaw,
  companyId: string,
  nowIso: string
): Warehouse {
  const id = String(raw.id || '').trim() || `wh-${companyId}-${String(raw.code || 'x')}`;
  const nameAr =
    String(raw.nameAr || raw.name_ar || raw.name || raw.code || id).trim() || id;
  return {
    id,
    tenantId: companyId,
    companyId,
    code: String(raw.code || id).slice(0, 64),
    nameAr,
    nameEn: raw.nameEn ? String(raw.nameEn) : undefined,
    location: raw.location ? String(raw.location) : undefined,
    valuationMethod: 'WA',
    isActive: raw.isActive !== false && raw.is_active !== false,
    createdAt: nowIso,
    updatedAt: nowIso,
  };
}

export interface WarehouseMigrateResult {
  ok: boolean;
  plane: 'an-migrate-write-f2';
  writesEnabled: boolean;
  writesPerformed: boolean;
  cutoverApproved: false;
  companyId: string;
  attempted: number;
  inserted: number;
  skippedExisting: number;
  failed: number;
  errors: string[];
  detail: string;
  detailAr: string;
}

export async function migrateWarehousesToAn(
  router: StorageRouter,
  opts: {
    companyId: string;
    rows: KemitWarehouseRaw[];
    env?: NodeJS.ProcessEnv;
  }
): Promise<WarehouseMigrateResult> {
  const env = opts.env || process.env;
  const companyId = opts.companyId.trim() || 'default';
  const writesEnabled =
    env.AN_MIGRATE_WRITES === '1' || env.AN_MIGRATE_WRITES === 'true';

  const base = {
    plane: 'an-migrate-write-f2' as const,
    writesEnabled,
    cutoverApproved: false as const,
    companyId,
  };

  if (!writesEnabled) {
    return {
      ...base,
      ok: false,
      writesPerformed: false,
      attempted: 0,
      inserted: 0,
      skippedExisting: 0,
      failed: 0,
      errors: ['AN_MIGRATE_WRITES not enabled'],
      detail: 'writes_disabled',
      detailAr: 'الكتابة معطّلة — اضبط AN_MIGRATE_WRITES=1',
    };
  }

  const plane = probeDefaultProcessPlane(env);
  if (!(plane.adapterType === 'postgresql' && plane.ok && plane.envRefPresent)) {
    return {
      ...base,
      ok: false,
      writesPerformed: false,
      attempted: 0,
      inserted: 0,
      skippedExisting: 0,
      failed: 0,
      errors: ['postgresql env not ready'],
      detail: plane.certifiedNote,
      detailAr: plane.errorAr || 'مستوى PostgreSQL غير جاهز',
    };
  }

  const adapter = router.getPrimaryAdapter();
  const existing = await adapter.listWarehouses(companyId);
  const existingIds = new Set(
    existing.success && Array.isArray(existing.data)
      ? existing.data.map((w) => w.id)
      : []
  );

  const nowIso = new Date().toISOString();
  let inserted = 0;
  let skippedExisting = 0;
  let failed = 0;
  const errors: string[] = [];

  for (const raw of opts.rows) {
    const mapped = mapKemitWarehouseToAn(raw, companyId, nowIso);
    if (existingIds.has(mapped.id)) {
      skippedExisting++;
      continue;
    }
    try {
      const key = `migrate-wh-${mapped.id}`;
      const res = await adapter.saveWarehouse(companyId, mapped, key);
      if (res.success) {
        inserted++;
        existingIds.add(mapped.id);
      } else {
        failed++;
        errors.push(`${mapped.id}:${res.error || 'save_failed'}`);
      }
    } catch (e: any) {
      failed++;
      errors.push(`${mapped.id}:${(e?.message || String(e)).slice(0, 120)}`);
    }
  }

  return {
    ...base,
    ok: failed === 0,
    writesPerformed: inserted > 0,
    attempted: opts.rows.length,
    inserted,
    skippedExisting,
    failed,
    errors: errors.slice(0, 20),
    detail: 'warehouse_migrate_done',
    detailAr: 'انتهى ترحيل المستودعات (إن وُجدت صفوف)',
  };
}
```

If `valuationMethod: 'WA'` is invalid, use the union value already used in codebase (check InventoryValuationMethod — common: 'weighted_average' | 'WA' | 'FIFO'). Adjust to a valid enum from StoragePort.

## STEP 3b — PATCH `server.ts`
```javascript
    app.post("/api/an/migrate/inventory/warehouses", async (req: any, res: any) => {
      try {
        const { migrateWarehousesToAn } = await import("./src/lib/anMigrateInventoryWrite");
        const companyId = String(req.body?.companyId || req.query?.companyId || "").trim();
        if (!companyId) {
          return res.status(400).json({ ok: false, error: "companyId required" });
        }
        let rows = Array.isArray(req.body?.rows) ? req.body.rows : null;
        if (!rows) {
          // Load from Firestore when adminDb available
          rows = [];
          if (typeof adminDb !== "undefined" && adminDb) {
            try {
              const snap = await adminDb.collection("warehouses").where("company_id", "==", companyId).get();
              rows = snap.docs.map((d: any) => ({ id: d.id, ...d.data() }));
            } catch (e: any) {
              try {
                const snap2 = await adminDb.collection("warehouses").limit(500).get();
                rows = snap2.docs
                  .map((d: any) => ({ id: d.id, ...d.data() }))
                  .filter((r: any) => r.company_id === companyId);
              } catch (e2: any) {
                return res.status(500).json({
                  ok: false,
                  error: "failed_to_load_kemit_warehouses",
                  detail: e2?.message || e?.message,
                });
              }
            }
          }
        }
        const result = await migrateWarehousesToAn(anStorageRouter, {
          companyId,
          rows: rows || [],
        });
        const status = result.writesEnabled ? 200 : 403;
        res.status(status).json(result);
      } catch (err: any) {
        res.status(500).json({ ok: false, error: err?.message || String(err) });
      }
    });
```
