#### 4.1 `src/lib/anBridge.ts`
```typescript
/**
 * @file anBridge.ts
 * @description Starter bridge between CustodyFlow UI and A.N core (src/an-core).
 * Client-safe: imports ONLY types from an-core (no server code in the browser).
 * Later phases wire dbSetDoc / login through here. Phase 1 = probe only.
 */
import type {
  AdapterType,
  BackendConfig,
  IsolationLevel,
  StorageResult,
  TenantCompany,
} from '../an-core/storage/StoragePort';

export type { AdapterType, BackendConfig, IsolationLevel, StorageResult, TenantCompany };

export type AuthDriver = 'firebase' | 'an-jwt';

export const AN_CORE_VERSION = '7.2.0';

export interface CompanyBackendSettings {
  auth_driver: AuthDriver;
  db_backend: AdapterType;
}

/** Defaults keep existing Firebase path — A.N core is opt-in per company. */
export function getCompanyBackendSettings(
  company: { auth_driver?: AuthDriver; db_backend?: AdapterType } | null | undefined
): CompanyBackendSettings {
  return {
    auth_driver: company?.auth_driver ?? 'firebase',
    db_backend: company?.db_backend ?? 'firebase',
  };
}

/** Probes A.N core mounted on the server (GET /api/an/storage-status). */
export async function getAnStorageStatus(): Promise<{
  ok: boolean;
  adapterName?: string;
  tenants?: number;
  version?: string;
  error?: string;
}> {
  try {
    const res = await fetch('/api/an/storage-status');
    if (!res.ok) return { ok: false, error: `HTTP ${res.status}` };
    const data = await res.json();
    return { ok: true, ...data };
  } catch (e: any) {
    return { ok: false, error: e?.message || 'network error' };
  }
}
```


#### 4.2 `scripts/verifyMergeM1.ts`
```typescript
/**
 * verifyMergeM1.ts — MERGE phase 1 gates (core inject only)
 */
import fs from 'fs';
import path from 'path';

let passed = 0;
let failed = 0;
function check(name: string, cond: boolean, detail = '') {
  if (cond) {
    console.log(`[PASS] ${name}`);
    passed++;
  } else {
    console.log(`[FAIL] ${name}${detail ? ' — ' + detail : ''}`);
    failed++;
  }
}

const root = process.cwd();
const anCore = path.join(root, 'src/an-core');
const port = path.join(anCore, 'storage/StoragePort.ts');
const router = path.join(anCore, 'storage/StorageRouter.ts');
const bridge = path.join(root, 'src/lib/anBridge.ts');
const server = fs.readFileSync(path.join(root, 'server.ts'), 'utf8');
const pkg = JSON.parse(fs.readFileSync(path.join(root, 'package.json'), 'utf8'));
const types = fs.readFileSync(path.join(root, 'src/types.ts'), 'utf8');

check('S1 src/an-core exists', fs.existsSync(anCore));
check('S2 StoragePort.ts exists', fs.existsSync(port));
check('S3 StorageRouter.ts exists', fs.existsSync(router));
check('S4 anBridge.ts exists', fs.existsSync(bridge));
check('S5 anBridge exports getAnStorageStatus', fs.readFileSync(bridge, 'utf8').includes('getAnStorageStatus'));
check('S6 anBridge exports getCompanyBackendSettings', fs.readFileSync(bridge, 'utf8').includes('getCompanyBackendSettings'));
check('S7 server mounts /api/an/storage-status', server.includes("/api/an/storage-status"));
check('S8 server imports StorageRouter from an-core', /an-core\/storage\/StorageRouter/.test(server));
check('S9 package has pg', !!pkg.dependencies?.pg);
check('S10 package has argon2', !!pkg.dependencies?.argon2);
check('S11 types has db_backend', types.includes('db_backend'));
check('S12 types has auth_driver', types.includes('auth_driver'));
check('S13 firebase dependency kept', !!pkg.dependencies?.firebase);
check('S14 Login.tsx still present', fs.existsSync(path.join(root, 'src/pages/Login.tsx')));

console.log(`Passed: ${passed} | Failed: ${failed}`);
process.exit(failed > 0 ? 1 : 0);
```
