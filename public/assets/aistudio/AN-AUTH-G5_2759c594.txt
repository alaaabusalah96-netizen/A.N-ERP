# AN-AUTH-G5 — Wire platform admin bootstrap on server startup (LEAN)
# Root cause (Studio diagnosis): PLATFORM_ADMIN_* secrets present but bootstrapPlatformAdmin never called;
# zero users on tenant default → A.N login invalid_credentials.
# This package calls bootstrap once at startup on the same primary adapter used by /api/an/auth/login.

## HARD RULES
1. Never log password or JWT_SECRET values.
2. Do not change Login.tsx behavior rules from G4.
3. Bootstrap only when PLATFORM_ADMIN_USERNAME and PLATFORM_ADMIN_PASSWORD are both non-empty.
4. Optional gate: if AN_BOOTSTRAP_PLATFORM_ADMIN is explicitly `0` or `false`, skip (default = run when secrets present).
5. Failures log and continue server start (do not crash the whole process) unless you already have a fail-closed pattern — prefer log + continue so Preview stays up.
6. Stop at first package-apply failure. tsc wait >2 min → once → build → 9–12. One RESULTS REPORT only.

## STEP 3 — Create `src/lib/anPlatformBootstrap.ts`

```typescript
/**
 * AN-AUTH-G5 — run platform admin bootstrap against A.N primary storage.
 */
import type { StorageRouter } from '../an-core/storage/StorageRouter';
import { bootstrapPlatformAdmin } from '../an-core/auth/platformAdminBootstrap';

export function shouldBootstrapPlatformAdmin(env: NodeJS.ProcessEnv = process.env): boolean {
  if (env.AN_BOOTSTRAP_PLATFORM_ADMIN === '0' || env.AN_BOOTSTRAP_PLATFORM_ADMIN === 'false') {
    return false;
  }
  const u = (env.PLATFORM_ADMIN_USERNAME || '').trim();
  const p = env.PLATFORM_ADMIN_PASSWORD || '';
  return u.length > 0 && p.length > 0;
}

export async function runPlatformAdminBootstrapOnStart(
  router: StorageRouter,
  env: NodeJS.ProcessEnv = process.env
): Promise<{ ran: boolean; outcome?: string; code?: string; message?: string }> {
  if (!shouldBootstrapPlatformAdmin(env)) {
    return { ran: false, outcome: 'skipped_no_secrets_or_flag_off' };
  }

  const tenantId = (env.PLATFORM_ADMIN_TENANT_ID || 'default').trim() || 'default';
  const username = (env.PLATFORM_ADMIN_USERNAME || '').trim();
  const password = env.PLATFORM_ADMIN_PASSWORD || '';

  const adapter = router.getPrimaryAdapter();

  // Ensure PG schema if adapter supports it
  try {
    if (typeof (adapter as any).initializeSchema === 'function') {
      const init = await (adapter as any).initializeSchema();
      if (init && init.success === false) {
        return {
          ran: true,
          outcome: 'schema_init_failed',
          code: 'schema_init_failed',
          message: init.error?.message || 'initializeSchema failed',
        };
      }
    }
  } catch (e: any) {
    return {
      ran: true,
      outcome: 'schema_init_threw',
      code: 'schema_init_threw',
      message: (e?.message || String(e)).slice(0, 200),
    };
  }

  try {
    const result = await bootstrapPlatformAdmin(adapter, tenantId, username, password);
    if (result.ok) {
      return { ran: true, outcome: result.outcome };
    }
    return {
      ran: true,
      outcome: 'failed',
      code: result.code,
      message: result.message,
    };
  } catch (e: any) {
    return {
      ran: true,
      outcome: 'threw',
      code: 'bootstrap_threw',
      message: (e?.message || String(e)).slice(0, 200),
    };
  }
}
```

## STEP 3b — PATCH `server.ts`
Inside the A.N core bridge `try` block, **after** `const anStorageRouter = new StorageRouter();` and **before** or **after** routes are registered (must run once per process start). Prefer **await once** before `app.listen`, still inside the same scope where `anStorageRouter` exists.

If `anStorageRouter` is block-scoped inside try and listen is outside, either:
- hoist bootstrap call at end of the A.N try (async IIFE), or
- store router on `app.locals.anStorageRouter` and bootstrap there.

Recommended pattern at end of A.N try (before closing the try):

```javascript
    try {
      const { runPlatformAdminBootstrapOnStart } = await import("./src/lib/anPlatformBootstrap");
      const boot = await runPlatformAdminBootstrapOnStart(anStorageRouter, process.env);
      if (!boot.ran) {
        console.log("[AN-AUTH-G5] platform bootstrap skipped:", boot.outcome);
      } else if (boot.outcome === "created" || boot.outcome === "already_exists" || boot.outcome === "migrated_synced" || boot.outcome === "migrated_created") {
        console.log("[AN-AUTH-G5] platform bootstrap:", boot.outcome);
      } else {
        console.warn("[AN-AUTH-G5] platform bootstrap issue:", boot.outcome, boot.code || "", boot.message || "");
      }
    } catch (bootErr: any) {
      console.warn("[AN-AUTH-G5] platform bootstrap error:", bootErr?.message || bootErr);
    }
```

**Critical:** `anStyleLogin` must use the **same** `anStorageRouter` instance (already does).
