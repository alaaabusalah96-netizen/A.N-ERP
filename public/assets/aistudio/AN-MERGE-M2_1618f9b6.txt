### 3.1 Overwrite `src/lib/anBridge.ts`
```typescript
/**
 * @file anBridge.ts
 * @description Bridge between CustodyFlow UI and A.N core (src/an-core).
 * Client-safe: imports ONLY types from an-core (no server adapters in the browser).
 * MERGE-M2: company backend binding (db_backend / isolation / connectionRef).
 */
import type {
  AdapterType,
  BackendConfig,
  IsolationLevel,
  StorageResult,
  TenantCompany,
} from '../an-core/storage/StoragePort';

export type { AdapterType, BackendConfig, IsolationLevel, StorageResult, TenantCompany };

export type AuthDriver = 'firebase' | 'an-jwt';

export const AN_CORE_VERSION = '7.2.0';

export const ALLOWED_DB_BACKENDS: AdapterType[] = [
  'firebase',
  'postgresql',
  'mysql',
  'mongodb',
  'inmemory',
];

export const ALLOWED_ISOLATION: IsolationLevel[] = ['Shared', 'Dedicated'];

export interface CompanyBackendSettings {
  auth_driver: AuthDriver;
  db_backend: AdapterType;
  isolation_level: IsolationLevel;
  /** Name of env/secret reference only — never raw credentials */
  connection_ref: string;
}

type CompanyLike = {
  id?: string;
  auth_driver?: AuthDriver;
  db_backend?: AdapterType;
  isolation_level?: IsolationLevel;
  connection_ref?: string;
} | null | undefined;

/** Defaults keep existing Firebase path — A.N SQL backends are opt-in per company. */
export function getCompanyBackendSettings(company: CompanyLike): CompanyBackendSettings {
  const db = company?.db_backend && ALLOWED_DB_BACKENDS.includes(company.db_backend)
    ? company.db_backend
    : 'firebase';
  const isolation =
    company?.isolation_level && ALLOWED_ISOLATION.includes(company.isolation_level)
      ? company.isolation_level
      : 'Shared';
  let connection_ref = (company?.connection_ref || '').trim();
  if (!connection_ref) {
    if (db === 'postgresql') connection_ref = 'DATABASE_URL';
    else if (db === 'mysql') connection_ref = 'MYSQL_URL';
    else if (db === 'mongodb') connection_ref = 'MONGODB_URL';
    else connection_ref = '';
  }
  return {
    auth_driver: company?.auth_driver === 'an-jwt' ? 'an-jwt' : 'firebase',
    db_backend: db,
    isolation_level: isolation,
    connection_ref,
  };
}

export function validateCompanyBackendSettings(
  settings: CompanyBackendSettings
): { valid: boolean; errorAr?: string; errorEn?: string } {
  if (!ALLOWED_DB_BACKENDS.includes(settings.db_backend)) {
    return {
      valid: false,
      errorAr: 'نوع قاعدة البيانات غير مدعوم',
      errorEn: 'Unsupported db_backend',
    };
  }
  if (!ALLOWED_ISOLATION.includes(settings.isolation_level)) {
    return {
      valid: false,
      errorAr: 'مستوى العزل غير صالح',
      errorEn: 'Invalid isolation_level',
    };
  }
  if (settings.auth_driver !== 'firebase' && settings.auth_driver !== 'an-jwt') {
    return {
      valid: false,
      errorAr: 'محرك المصادقة غير صالح',
      errorEn: 'Invalid auth_driver',
    };
  }
  const needsRef =
    settings.db_backend === 'postgresql' ||
    settings.db_backend === 'mysql' ||
    settings.db_backend === 'mongodb';
  if (needsRef) {
    const ref = (settings.connection_ref || '').trim();
    if (!ref) {
      return {
        valid: false,
        errorAr: 'يجب تحديد اسم مرجع الاتصال (مثل DATABASE_URL) وليس كلمة السر نفسها',
        errorEn: 'connection_ref (env name) is required for SQL/document backends',
      };
    }
    if (ref.includes('://') || ref.includes('password') || ref.length > 128) {
      return {
        valid: false,
        errorAr: 'connection_ref يجب أن يكون اسم متغير بيئة فقط وليس سلسلة اتصال كاملة',
        errorEn: 'connection_ref must be an env var name, not a raw connection string',
      };
    }
  }
  return { valid: true };
}

/** Maps company settings to A.N StoragePort BackendConfig (server-side use). */
export function companyToBackendConfig(company: CompanyLike): BackendConfig {
  const s = getCompanyBackendSettings(company);
  return {
    adapterType: s.db_backend,
    connectionRef: s.connection_ref || 'DATABASE_URL',
    isolationLevel: s.isolation_level,
  };
}

/** Probes A.N core mounted on the server (GET /api/an/storage-status). */
export async function getAnStorageStatus(): Promise<{
  ok: boolean;
  adapterName?: string;
  tenants?: number;
  version?: string;
  error?: string;
}> {
  try {
    const res = await fetch('/api/an/storage-status');
    if (!res.ok) return { ok: false, error: `HTTP ${res.status}` };
    const data = await res.json();
    return { ok: true, ...data };
  } catch (e: any) {
    return { ok: false, error: e?.message || 'network error' };
  }
}

/** Resolve backend binding for a company id (server validates + echoes config). */
export async function resolveCompanyBackend(
  companyId: string,
  partial?: Partial<CompanyBackendSettings>
): Promise<{ ok: boolean; settings?: CompanyBackendSettings; backendConfig?: BackendConfig; error?: string }> {
  try {
    const res = await fetch(`/api/an/company-backend/${encodeURIComponent(companyId)}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      credentials: 'include',
      body: JSON.stringify(partial || {}),
    });
    const data = await res.json().catch(() => ({}));
    if (!res.ok) return { ok: false, error: data?.error || `HTTP ${res.status}` };
    return { ok: true, ...data };
  } catch (e: any) {
    return { ok: false, error: e?.message || 'network error' };
  }
}
```

### 3.2 Create `src/components/settings/CompanyBackendPanel.tsx`
```tsx
/**
 * CompanyBackendPanel — MERGE-M2
 * Lets Admin/SystemAdmin set per-company db_backend + isolation + connection_ref.
 * Persists via CustodyDB.updateCompany. Does NOT switch live traffic off Firebase
 * until later merge phases; it stores the binding contract only.
 */
import React, { useEffect, useMemo, useState } from 'react';
import { Database, Save, AlertTriangle, CheckCircle2 } from 'lucide-react';
import { Company, Language } from '../../types';
import { CustodyDB } from '../../db';
import {
  ALLOWED_DB_BACKENDS,
  ALLOWED_ISOLATION,
  CompanyBackendSettings,
  getCompanyBackendSettings,
  validateCompanyBackendSettings,
  resolveCompanyBackend,
  type AdapterType,
  type IsolationLevel,
  type AuthDriver,
} from '../../lib/anBridge';

interface Props {
  language: Language;
  /** When empty, uses selected company from CustodyDB */
  companyId?: string;
}

export default function CompanyBackendPanel({ language, companyId }: Props) {
  const isRtl = language === 'ar';
  const companies = CustodyDB.getCompanies();
  const selectedId =
    companyId ||
    CustodyDB.getSelectedCompanyId() ||
    (companies[0] && companies[0].id) ||
    '';
  const company: Company | undefined = companies.find((c) => c.id === selectedId);

  const initial = useMemo(() => getCompanyBackendSettings(company), [company?.id, company?.db_backend, company?.auth_driver, company?.isolation_level, company?.connection_ref]);
  const [authDriver, setAuthDriver] = useState<AuthDriver>(initial.auth_driver);
  const [dbBackend, setDbBackend] = useState<AdapterType>(initial.db_backend);
  const [isolation, setIsolation] = useState<IsolationLevel>(initial.isolation_level);
  const [connectionRef, setConnectionRef] = useState(initial.connection_ref);
  const [msg, setMsg] = useState<{ type: 'ok' | 'err'; text: string } | null>(null);
  const [busy, setBusy] = useState(false);

  useEffect(() => {
    const s = getCompanyBackendSettings(company);
    setAuthDriver(s.auth_driver);
    setDbBackend(s.db_backend);
    setIsolation(s.isolation_level);
    setConnectionRef(s.connection_ref);
    setMsg(null);
  }, [selectedId]);

  if (!company) {
    return (
      <div className="rounded-2xl border border-slate-200 dark:border-slate-700 bg-white/60 dark:bg-slate-900/40 p-4 text-sm text-slate-500">
        {isRtl ? 'لا توجد شركة محددة لربط قاعدة البيانات.' : 'No company selected for backend binding.'}
      </div>
    );
  }

  async function handleSave(e: React.FormEvent) {
    e.preventDefault();
    setMsg(null);
    const settings: CompanyBackendSettings = {
      auth_driver: authDriver,
      db_backend: dbBackend,
      isolation_level: isolation,
      connection_ref: connectionRef.trim(),
    };
    const v = validateCompanyBackendSettings(settings);
    if (!v.valid) {
      setMsg({ type: 'err', text: isRtl ? (v.errorAr || 'خطأ') : (v.errorEn || 'Invalid') });
      return;
    }
    setBusy(true);
    try {
      const server = await resolveCompanyBackend(company!.id, settings);
      if (!server.ok) {
        setMsg({
          type: 'err',
          text: isRtl
            ? `رفض الخادم الإعداد: ${server.error || 'unknown'}`
            : `Server rejected settings: ${server.error || 'unknown'}`,
        });
        setBusy(false);
        return;
      }
      const res = await CustodyDB.updateCompany(company!.id, {
        auth_driver: settings.auth_driver,
        db_backend: settings.db_backend,
        isolation_level: settings.isolation_level,
        connection_ref: settings.connection_ref,
      });
      if (!res.success) {
        setMsg({
          type: 'err',
          text: res.error || (isRtl ? 'فشل الحفظ' : 'Save failed'),
        });
      } else {
        setMsg({
          type: 'ok',
          text: isRtl
            ? 'تم حفظ ربط قاعدة البيانات للشركة (الافتراضي firebase يبقى حتى تفعيل التشغيل لاحقاً).'
            : 'Company backend binding saved (runtime stays on Firebase until a later phase enables the adapter).',
        });
      }
    } catch (err: any) {
      setMsg({ type: 'err', text: err?.message || String(err) });
    }
    setBusy(false);
  }

  const needsRef = dbBackend === 'postgresql' || dbBackend === 'mysql' || dbBackend === 'mongodb';

  return (
    <div className="rounded-2xl border border-indigo-500/20 bg-white/70 dark:bg-slate-900/50 p-4 md:col-span-2">
      <div className="flex items-center gap-2 mb-3">
        <div className="h-9 w-9 rounded-xl bg-indigo-500/15 text-indigo-500 flex items-center justify-center">
          <Database size={18} />
        </div>
        <div>
          <h3 className="font-bold text-slate-800 dark:text-slate-100 text-sm">
            {isRtl ? 'ربط قاعدة بيانات الشركة' : 'Company database binding'}
          </h3>
          <p className="text-[11px] text-slate-500">
            {company.name} · <span className="font-mono">{company.id}</span>
          </p>
        </div>
      </div>

      <form onSubmit={handleSave} className="grid gap-3 md:grid-cols-2">
        <label className="block text-xs font-bold text-slate-500">
          {isRtl ? 'محرك المصادقة' : 'Auth driver'}
          <select
            className="mt-1 w-full rounded-xl border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-950 px-3 py-2 text-sm"
            value={authDriver}
            onChange={(e) => setAuthDriver(e.target.value as AuthDriver)}
          >
            <option value="firebase">firebase (default)</option>
            <option value="an-jwt">an-jwt (A.N core — later phases)</option>
          </select>
        </label>

        <label className="block text-xs font-bold text-slate-500">
          {isRtl ? 'نوع قاعدة البيانات' : 'Database backend'}
          <select
            className="mt-1 w-full rounded-xl border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-950 px-3 py-2 text-sm"
            value={dbBackend}
            onChange={(e) => setDbBackend(e.target.value as AdapterType)}
          >
            {ALLOWED_DB_BACKENDS.map((b) => (
              <option key={b} value={b}>
                {b}
              </option>
            ))}
          </select>
        </label>

        <label className="block text-xs font-bold text-slate-500">
          {isRtl ? 'مستوى العزل' : 'Isolation level'}
          <select
            className="mt-1 w-full rounded-xl border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-950 px-3 py-2 text-sm"
            value={isolation}
            onChange={(e) => setIsolation(e.target.value as IsolationLevel)}
          >
            {ALLOWED_ISOLATION.map((x) => (
              <option key={x} value={x}>
                {x}
              </option>
            ))}
          </select>
        </label>

        <label className="block text-xs font-bold text-slate-500">
          {isRtl ? 'مرجع الاتصال (اسم متغير البيئة فقط)' : 'Connection ref (env var name only)'}
          <input
            className="mt-1 w-full rounded-xl border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-950 px-3 py-2 text-sm font-mono"
            value={connectionRef}
            disabled={!needsRef}
            placeholder={needsRef ? 'DATABASE_URL' : '—'}
            onChange={(e) => setConnectionRef(e.target.value)}
          />
        </label>

        <div className="md:col-span-2 flex flex-wrap items-center gap-3">
          <button
            type="submit"
            disabled={busy}
            className="inline-flex items-center gap-2 rounded-xl bg-indigo-600 hover:bg-indigo-500 text-white text-sm font-bold px-4 py-2 disabled:opacity-50"
          >
            <Save size={16} />
            {isRtl ? 'حفظ الربط' : 'Save binding'}
          </button>
          {msg && (
            <span
              className={`inline-flex items-center gap-1.5 text-xs font-semibold ${
                msg.type === 'ok' ? 'text-emerald-600' : 'text-rose-500'
              }`}
            >
              {msg.type === 'ok' ? <CheckCircle2 size={14} /> : <AlertTriangle size={14} />}
              {msg.text}
            </span>
          )}
        </div>
      </form>
    </div>
  );
}
```
