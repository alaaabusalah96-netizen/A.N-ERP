## STEP 3a — PATCH src/App.tsx (platform-status timeout)
Replace the useEffect that fetches `/api/auth/platform-status` with:

```tsx
  useEffect(() => {
    if (activeUser) return;
    let isMounted = true;
    const timer = window.setTimeout(() => {
      if (isMounted && hasPlatformOwner === null) {
        console.warn('platform-status timeout → Login');
        setHasPlatformOwner(true);
      }
    }, 4000);
    fetch('/api/auth/platform-status')
      .then((res) => {
        if (!res.ok) throw new Error('Status endpoint returned non-200');
        return res.json();
      })
      .then((data) => {
        if (isMounted) setHasPlatformOwner(data?.hasPlatformOwner ?? true);
      })
      .catch((err) => {
        console.warn('Failed to check platform status, defaulting to Login screen:', err);
        if (isMounted) setHasPlatformOwner(true);
      })
      .finally(() => {
        window.clearTimeout(timer);
      });
    return () => {
      isMounted = false;
      window.clearTimeout(timer);
    };
  }, [activeUser]);
```

## STEP 3c — Error boundary (minimal)
Create `src/components/BootErrorBoundary.tsx`:

```tsx
import React from 'react';

type Props = { children: React.ReactNode };
type State = { error: Error | null };

export default class BootErrorBoundary extends React.Component<Props, State> {
  state: State = { error: null };
  static getDerivedStateFromError(error: Error) {
    return { error };
  }
  render() {
    if (this.state.error) {
      return (
        <div style={{ padding: 24, fontFamily: 'sans-serif', direction: 'rtl' }}>
          <h1>خطأ في تحميل الواجهة</h1>
          <pre style={{ whiteSpace: 'pre-wrap', color: '#b91c1c' }}>
            {this.state.error.message}
          </pre>
          <button type="button" onClick={() => window.location.reload()}>
            إعادة التحميل
          </button>
        </div>
      );
    }
    return this.props.children;
  }
}
```

In `src/main.tsx` (or `src/index.tsx`), wrap `<App />` with `<BootErrorBoundary>`.
