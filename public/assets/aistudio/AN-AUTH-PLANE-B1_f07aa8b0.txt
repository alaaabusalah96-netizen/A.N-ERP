## STEP 3 — Create `src/lib/anSessionPlane.ts`

```typescript
/**
 * AN-AUTH-PLANE-B1 — combine auth policy + company data-plane probe for UI/server callers.
 */
import { resolveAuthPolicyServer } from './authPolicy';
import {
  probeCompanyPlane,
  liveCheckBackend,
  type CompanyPlaneResult,
  type LiveCheckResult,
} from './anDataPlane';
import type { AuthPolicy } from './authPolicy';

export interface SessionPlaneResult {
  ok: boolean;
  companyId: string;
  auth: AuthPolicy;
  companyPlane: CompanyPlaneResult;
  live?: LiveCheckResult;
  loginPath: string;
  firestoreOperationalPath: 'unchanged';
}

export async function buildSessionPlane(input: {
  companyId: string;
  companyRecord?: Record<string, unknown> | null;
  companySource?: CompanyPlaneResult['companySource'];
  runLive?: boolean;
  env?: NodeJS.ProcessEnv;
}): Promise<SessionPlaneResult> {
  const env = input.env || process.env;
  const companyId = (input.companyId || '').trim() || 'unknown';
  const source = input.companySource || (input.companyRecord ? 'firestore' : 'process-default');
  const companyPlane = probeCompanyPlane(companyId, input.companyRecord || null, source, env);
  const auth = resolveAuthPolicyServer(
    { auth_driver: companyPlane.settings.auth_driver as 'firebase' | 'an-jwt' },
    env
  );
  let live: LiveCheckResult | undefined;
  if (input.runLive) {
    live = await liveCheckBackend(
      {
        db_backend: companyPlane.settings.db_backend as any,
        isolation_level: companyPlane.settings.isolation_level as any,
        connection_ref: companyPlane.settings.connection_ref,
        auth_driver: companyPlane.settings.auth_driver as any,
      },
      env
    );
  }
  return {
    ok: companyPlane.probe.ok && auth.activePath === 'firebase-session',
    companyId,
    auth,
    companyPlane,
    live,
    loginPath: '/api/auth/verify-credentials',
    firestoreOperationalPath: 'unchanged',
  };
}
```

    app.get("/api/an/session-plane", async (req: any, res: any) => {
      try {
        const { buildSessionPlane } = await import("./src/lib/anSessionPlane");
        const companyId = typeof req.query.companyId === "string" ? req.query.companyId.trim() : "";
        const runLive = req.query.live === "1" || req.query.live === "true";
        let record: Record<string, unknown> | null = null;
        let companySource: "firestore" | "process-default" | "unavailable" = "process-default";
        if (companyId) {
          try {
            if (typeof adminDb !== "undefined" && adminDb) {
              const snap = await adminDb.collection("companies").doc(companyId).get();
              if (snap.exists) {
                record = snap.data() as Record<string, unknown>;
                companySource = "firestore";
              } else {
                companySource = "firestore";
                record = null;
              }
            } else {
              companySource = "unavailable";
            }
          } catch (e: any) {
            console.warn("session-plane company read:", e?.message || e);
            companySource = "unavailable";
          }
        }
        const result = await buildSessionPlane({
          companyId: companyId || "process",
          companyRecord: record,
          companySource: companyId ? companySource : "process-default",
          runLive,
        });
        res.json({
          ok: result.ok,
          plane: "an-auth-plane-b1",
          companyId: result.companyId,
          auth: result.auth,
          settings: result.companyPlane.settings,
          probe: {
            ok: result.companyPlane.probe.ok,
            adapterType: result.companyPlane.probe.adapterType,
            isolationLevel: result.companyPlane.probe.isolationLevel,
            connectionRef: result.companyPlane.probe.connectionRef,
            envRefPresent: result.companyPlane.probe.envRefPresent,
            requiresEnvRef: result.companyPlane.probe.requiresEnvRef,
            certifiedNote: result.companyPlane.probe.certifiedNote,
            errorEn: result.companyPlane.probe.errorEn,
          },
          live: result.live
            ? {
                attempted: result.live.attempted,
                skipped: result.live.skipped,
                ok: result.live.ok,
                adapterType: result.live.adapterType,
                connectionRef: result.live.connectionRef,
                latencyMs: result.live.latencyMs,
                detail: result.live.detail,
              }
            : null,
          loginPath: result.loginPath,
          firestoreOperationalPath: result.firestoreOperationalPath,
        });
      } catch (err: any) {
        res.status(500).json({ ok: false, error: err?.message || String(err) });
      }
    });
