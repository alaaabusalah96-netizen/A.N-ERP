```typescript
export interface CompanyPlaneResult {
  companyId: string;
  companyFound: boolean;
  companySource: 'firestore' | 'request-body' | 'process-default' | 'unavailable';
  settings: {
    auth_driver: string;
    db_backend: string;
    isolation_level: string;
    connection_ref: string;
  };
  probe: DataPlaneProbe;
}

/** Map a loose company record (Firestore doc / body) into probe settings. */
export function companyRecordToPartial(record: Record<string, unknown> | null | undefined): Partial<CompanyBackendSettings> {
  if (!record) return {};
  const out: Partial<CompanyBackendSettings> = {};
  const auth = record.auth_driver;
  const db = record.db_backend;
  const iso = record.isolation_level;
  const ref = record.connection_ref;
  if (auth === 'firebase' || auth === 'an-jwt') out.auth_driver = auth;
  if (
    db === 'postgresql' ||
    db === 'mysql' ||
    db === 'mongodb' ||
    db === 'firebase' ||
    db === 'inmemory'
  ) {
    out.db_backend = db;
  }
  if (iso === 'Shared' || iso === 'Dedicated') out.isolation_level = iso;
  if (typeof ref === 'string') out.connection_ref = ref;
  return out;
}

export function probeCompanyPlane(
  companyId: string,
  record: Record<string, unknown> | null | undefined,
  source: CompanyPlaneResult['companySource'],
  env: NodeJS.ProcessEnv = process.env
): CompanyPlaneResult {
  const partial = companyRecordToPartial(record);
  const hasBinding =
    partial.db_backend != null ||
    partial.connection_ref != null ||
    partial.isolation_level != null ||
    partial.auth_driver != null;
  const probe = hasBinding
    ? probeBackendBinding(partial, env)
    : probeDefaultProcessPlane(env);
  const settings = getCompanyBackendSettings({
    id: companyId,
    ...partial,
  } as any);
  return {
    companyId,
    companyFound: source === 'firestore' || source === 'request-body',
    companySource: hasBinding ? source : source === 'firestore' && !hasBinding ? 'firestore' : source,
    settings: {
      auth_driver: settings.auth_driver,
      db_backend: settings.db_backend,
      isolation_level: settings.isolation_level,
      connection_ref: settings.connection_ref,
    },
    probe,
  };
}
```

```javascript
    app.get("/api/an/data-plane/company/:companyId", async (req: any, res: any) => {
      try {
        const companyId = String(req.params.companyId || "").trim();
        if (!companyId) {
          return res.status(400).json({ ok: false, error: "companyId required" });
        }
        const { probeCompanyPlane } = await import("./src/lib/anDataPlane");
        let record: Record<string, unknown> | null = null;
        let source: "firestore" | "unavailable" = "unavailable";
        try {
          if (typeof adminDb !== "undefined" && adminDb) {
            const snap = await adminDb.collection("companies").doc(companyId).get();
            if (snap.exists) {
              record = snap.data() as Record<string, unknown>;
              source = "firestore";
            } else {
              source = "firestore";
              record = null;
            }
          }
        } catch (readErr: any) {
          console.warn("data-plane company read:", readErr?.message || readErr);
          source = "unavailable";
          record = null;
        }
        const result = probeCompanyPlane(companyId, record, source === "unavailable" ? "unavailable" : "firestore");
        res.json({
          ok: result.probe.ok,
          plane: "an-data-plane-a2",
          companyId: result.companyId,
          companyFound: !!(record && source === "firestore"),
          companySource: result.companySource,
          settings: result.settings,
          probe: {
            adapterType: result.probe.adapterType,
            isolationLevel: result.probe.isolationLevel,
            connectionRef: result.probe.connectionRef,
            envRefPresent: result.probe.envRefPresent,
            requiresEnvRef: result.probe.requiresEnvRef,
            certifiedNote: result.probe.certifiedNote,
            errorAr: result.probe.errorAr,
            errorEn: result.probe.errorEn,
          },
          firestoreOperationalPath: "unchanged",
        });
      } catch (err: any) {
        res.status(500).json({ ok: false, error: err?.message || String(err) });
      }
    });
```
