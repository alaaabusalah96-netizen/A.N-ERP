# AN-DEV-PREVIEW-BYPASS-1 — Dev-only full-admin preview entry (LEAN)
# Goal: enter the app in Preview without real user/password while secrets are not ready.
# Creates a local SystemAdmin session with broad UI permissions when AN_PREVIEW_BYPASS=1.
# HARD: blocked when STRICT_PROD_SECURITY=true unless AN_PREVIEW_BYPASS_FORCE=1 (discouraged).

## HARD RULES
1. Files: `src/lib/anPreviewBypass.ts` (new), `src/pages/Login.tsx`, `server.ts` (`GET /api/an/auth/preview-bypass-status` only), `scripts/verifyAnDevPreviewBypass1.ts`, `package.json` one script, `.env.example`, `docs/MERGE_STATUS.md`.
2. Default OFF: bypass UI hidden unless server says enabled.
3. Preview user MUST be `role: 'SystemAdmin'`, `is_platform_owner: true`, rich `permissions` so sidebars/screens that check Admin/SystemAdmin open.
4. Do NOT accept arbitrary role from the client for bypass — fixed server/client template only.
5. Do NOT write production users to Postgres/Firestore from this package.
6. Never enable bypass in production docs as recommended; document as dev-only.
7. Stop at first failure. tsc >2 min → once → build → verify. One RESULTS REPORT only.

## PRECHECK
- Login.tsx uses onLoginSuccess(AppUser)
- AppUser has role, permissions, is_platform_owner
- cookie-parser / fetch patterns exist

## STEP 3 — `src/lib/anPreviewBypass.ts`

```typescript
/**
 * AN-DEV-PREVIEW-BYPASS-1 — local full-admin AppUser for UI preview only.
 */
import type { AppUser } from '../types';

export const PREVIEW_BYPASS_USER_ID = 'dev-preview-admin';

/** Broad UI permissions: keys used across KEMIT gates; unknown keys are harmless. */
export function buildPreviewAdminPermissions(): Record<string, boolean> {
  return {
    platform: true,
    admin: true,
    accounting: true,
    audit: true,
    inventory: true,
    purchasing: true,
    sales: true,
    hr: true,
    custody: true,
    reports: true,
    settings: true,
    users: true,
    companies: true,
    warehouses: true,
    all: true,
  };
}

export function buildPreviewAdminUser(language: 'ar' | 'en' = 'ar'): AppUser {
  return {
    id: PREVIEW_BYPASS_USER_ID,
    email: 'preview-admin@local.dev',
    full_name: language === 'ar' ? 'مدير معاينة (تطوير)' : 'Preview Admin (dev)',
    name: language === 'ar' ? 'معاينة' : 'Preview',
    role: 'SystemAdmin',
    userRoles: ['SystemAdmin', 'Admin'],
    assigned_roles: ['SystemAdmin'],
    permissions: buildPreviewAdminPermissions() as any,
    is_active: true,
    is_platform_owner: true,
    must_change_password: false,
    company_id: undefined,
    tenantId: 'default',
  };
}

export function isPreviewBypassEnvEnabled(env: {
  AN_PREVIEW_BYPASS?: string;
  AN_PREVIEW_BYPASS_FORCE?: string;
  STRICT_PROD_SECURITY?: string;
}): boolean {
  const on =
    env.AN_PREVIEW_BYPASS === '1' ||
    env.AN_PREVIEW_BYPASS === 'true';
  if (!on) return false;
  const strict =
    String(env.STRICT_PROD_SECURITY || '').toLowerCase() === 'true';
  if (strict) {
    return env.AN_PREVIEW_BYPASS_FORCE === '1' || env.AN_PREVIEW_BYPASS_FORCE === 'true';
  }
  return true;
}
```

## STEP 3b — `server.ts`
Near other `/api/an/auth/*`:

```javascript
    app.get("/api/an/auth/preview-bypass-status", (_req: any, res: any) => {
      const { isPreviewBypassEnvEnabled } = require("./src/lib/anPreviewBypass");
      // prefer dynamic import if project is ESM; match local style
      res.status(200).json({
        ok: true,
        enabled: isPreviewBypassEnvEnabled(process.env),
        role: "SystemAdmin",
        note: "dev only",
      });
    });
```
Use `await import("./src/lib/anPreviewBypass")` if the file is TS ESM like the rest of server.

Do **not** add a POST that issues real JWT unless you also want API access without secrets — UI-only bypass is enough for “see screens”. APIs may still 401 without secrets; that is acceptable for layout preview.
