## STEP 3 — Create `src/lib/anInventoryLists.ts`

```typescript
/**
 * AN-INV-READ-LIST-1 — list warehouses/items from resolved company adapter.
 */
import type { StorageRouter } from '../an-core/storage/StorageRouter';
import type { StoragePort, Warehouse, StockItem } from '../an-core/storage/StoragePort';
import type { CompanyAdapterResolution } from './anResolveCompanyAdapter';

export interface InventoryListResult<T> {
  ok: boolean;
  companyId: string;
  count: number;
  items: T[];
  adapterResolution?: Pick<
    CompanyAdapterResolution,
    'source' | 'adapterType' | 'isolationLevel' | 'connectionRef'
  >;
  error?: string;
}

export async function listAnWarehouses(
  adapter: StoragePort,
  companyId: string,
  resolution?: CompanyAdapterResolution
): Promise<InventoryListResult<Pick<Warehouse, 'id' | 'code' | 'nameAr' | 'isActive'>>> {
  try {
    const res = await adapter.listWarehouses(companyId, { isActive: undefined });
    if (!res.success) {
      return {
        ok: false,
        companyId,
        count: 0,
        items: [],
        error: res.error || 'listWarehouses failed',
        adapterResolution: resolution
          ? {
              source: resolution.source,
              adapterType: resolution.adapterType,
              isolationLevel: resolution.isolationLevel,
              connectionRef: resolution.connectionRef,
            }
          : undefined,
      };
    }
    const items = (res.data || []).map((w) => ({
      id: w.id,
      code: w.code,
      nameAr: w.nameAr,
      isActive: w.isActive,
    }));
    return {
      ok: true,
      companyId,
      count: items.length,
      items,
      adapterResolution: resolution
        ? {
            source: resolution.source,
            adapterType: resolution.adapterType,
            isolationLevel: resolution.isolationLevel,
            connectionRef: resolution.connectionRef,
          }
        : undefined,
    };
  } catch (e: any) {
    return {
      ok: false,
      companyId,
      count: 0,
      items: [],
      error: (e?.message || String(e)).slice(0, 200),
    };
  }
}

export async function listAnStockItems(
  adapter: StoragePort,
  companyId: string,
  resolution?: CompanyAdapterResolution
): Promise<InventoryListResult<Pick<StockItem, 'id' | 'itemCode' | 'nameAr' | 'isActive'>>> {
  try {
    const res = await adapter.listStockItems(companyId);
    if (!res.success) {
      return {
        ok: false,
        companyId,
        count: 0,
        items: [],
        error: res.error || 'listStockItems failed',
      };
    }
    const items = (res.data || []).map((it) => ({
      id: it.id,
      itemCode: it.itemCode,
      nameAr: it.nameAr,
      isActive: it.isActive,
    }));
    return {
      ok: true,
      companyId,
      count: items.length,
      items,
      adapterResolution: resolution
        ? {
            source: resolution.source,
            adapterType: resolution.adapterType,
            isolationLevel: resolution.isolationLevel,
            connectionRef: resolution.connectionRef,
          }
        : undefined,
    };
  } catch (e: any) {
    return {
      ok: false,
      companyId,
      count: 0,
      items: [],
      error: (e?.message || String(e)).slice(0, 200),
    };
  }
}
```

## STEP 3b — PATCH `server.ts`

```javascript
    app.get("/api/an/inventory/warehouses", async (req: any, res: any) => {
      try {
        const authResult = await authenticateSessionOrBearer(req);
        if (!authResult.authenticated) {
          return res.status(authResult.statusCode || 401).json({ ok: false, message: authResult.message });
        }
        const companyId = String(req.query.companyId || "").trim();
        if (!companyId) return res.status(400).json({ ok: false, error: "companyId required" });
        const { loadCompanyLikeFromFirestore, resolveCompanyStorageAdapter } = await import("./src/lib/anResolveCompanyAdapter");
        const { listAnWarehouses } = await import("./src/lib/anInventoryLists");
        const companyLike = await loadCompanyLikeFromFirestore(
          typeof adminDb !== "undefined" ? adminDb : null,
          companyId
        );
        let resolved;
        try {
          resolved = resolveCompanyStorageAdapter(anStorageRouter, companyId, companyLike, null);
        } catch (e: any) {
          return res.status(400).json({ ok: false, code: "adapter_resolve_failed", message: e?.message || String(e) });
        }
        const result = await listAnWarehouses(resolved.adapter, companyId, resolved);
        res.status(result.ok ? 200 : 500).json(result);
      } catch (err: any) {
        res.status(500).json({ ok: false, error: err?.message || String(err) });
      }
    });
```
