import fs from 'fs';

    passed++;
  } else {
    console.log(`[FAIL] ${name}${detail ? ' — ' + detail : ''}`);
    failed++;
  }
}

check('S5 displays auth path or live', panel.includes('authPath') || panel.includes('liveDetail'));

check('S7 still loads routes', panel.includes('/routes'));
