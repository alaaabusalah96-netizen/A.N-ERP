## STEP 3 — Create `src/lib/anMigrateInventoryDry.ts`

```typescript
/**
 * AN-MIGRATE-DRY-F1 — inventory migration dry-run report (no writes).
 */
import type { StorageRouter } from '../an-core/storage/StorageRouter';
import { readInventorySummary } from './anInventoryRead';
import { probeDefaultProcessPlane } from './anDataPlane';

export interface InventoryDryRunReport {
  ok: boolean;
  plane: 'an-migrate-dry-f1';
  dryRun: true;
  writesPerformed: false;
  cutoverApproved: false;
  companyId: string;
  processPlane: {
    adapterType: string;
    envRefPresent: boolean;
    ok: boolean;
  };
  an: {
    source: string;
    warehouseCount: number | null;
    itemCount: number | null;
  };
  kemit: {
    source: 'firestore-admin' | 'unavailable' | 'error';
    warehouseCount: number | null;
    productCount: number | null;
    detail?: string;
  };
  delta: {
    warehouses: number | null;
    productsVsItems: number | null;
  };
  plan: string[];
  planAr: string[];
}

/** Optional adminDb-like reader injected by server to avoid hard import cycles. */
export type KemitInventoryCounter = (companyId: string) => Promise<{
  warehouseCount: number | null;
  productCount: number | null;
  detail?: string;
}>;

export async function buildInventoryMigrateDryRun(
  router: StorageRouter,
  opts: {
    companyId?: string;
    env?: NodeJS.ProcessEnv;
    countKemit?: KemitInventoryCounter;
  }
): Promise<InventoryDryRunReport> {
  const env = opts.env || process.env;
  const companyId = (opts.companyId || 'default').trim() || 'default';
  const processPlane = probeDefaultProcessPlane(env);
  const anSummary = await readInventorySummary(router, { companyId, env });

  let kemit: InventoryDryRunReport['kemit'] = {
    source: 'unavailable',
    warehouseCount: null,
    productCount: null,
    detail: 'no_kemit_counter',
  };
  if (opts.countKemit) {
    try {
      const k = await opts.countKemit(companyId);
      kemit = {
        source: 'firestore-admin',
        warehouseCount: k.warehouseCount,
        productCount: k.productCount,
        detail: k.detail,
      };
    } catch (e: any) {
      kemit = {
        source: 'error',
        warehouseCount: null,
        productCount: null,
        detail: (e?.message || String(e)).slice(0, 160),
      };
    }
  }

  const whAn = anSummary.warehouseCount;
  const whKe = kemit.warehouseCount;
  const itAn = anSummary.itemCount;
  const prKe = kemit.productCount;

  const plan = [
    'Dry-run only: no SQL writes performed',
    'Next package (F2) may upsert warehouses when AN_MIGRATE_WRITES=1 and readiness gates allow',
    'Inventory.tsx remains on CustodyDB until explicit UI cutover package',
    'Reconcile company_id partitioning before any write migrate',
  ];
  const planAr = [
    'تجريبي فقط: لم تُكتب صفوف SQL',
    'الحزمة التالية (F2) قد ترفع المستودعات عند AN_MIGRATE_WRITES=1 وبعد بوابات الجاهزية',
    'Inventory.tsx يبقى على CustodyDB حتى حزمة قطع واجهة صريحة',
    'راجع عزل company_id قبل أي ترحيل كتابة',
  ];

  return {
    ok: true,
    plane: 'an-migrate-dry-f1',
    dryRun: true,
    writesPerformed: false,
    cutoverApproved: false,
    companyId,
    processPlane: {
      adapterType: processPlane.adapterType,
      envRefPresent: processPlane.envRefPresent,
      ok: processPlane.ok,
    },
    an: {
      source: anSummary.source,
      warehouseCount: whAn,
      itemCount: itAn,
    },
    kemit,
    delta: {
      warehouses:
        whAn != null && whKe != null ? whAn - whKe : null,
      productsVsItems:
        itAn != null && prKe != null ? itAn - prKe : null,
    },
    plan,
    planAr,
  };
}
```

## STEP 3b — PATCH `server.ts`
```javascript
    app.get("/api/an/migrate/inventory/dry-run", async (req: any, res: any) => {
      try {
        const { buildInventoryMigrateDryRun } = await import("./src/lib/anMigrateInventoryDry");
        const companyId = typeof req.query.companyId === "string" ? req.query.companyId.trim() : "default";
        const countKemit = async (cid: string) => {
          if (typeof adminDb === "undefined" || !adminDb) {
            return { warehouseCount: null, productCount: null, detail: "adminDb_unavailable" };
          }
          // Best-effort counts; collection names may vary — never throw out
          let warehouseCount: number | null = null;
          let productCount: number | null = null;
          const details: string[] = [];
          try {
            const wh = await adminDb.collection("warehouses").where("company_id", "==", cid).get();
            warehouseCount = wh.size;
          } catch (e: any) {
            details.push("warehouses:" + (e?.message || "err"));
            try {
              const wh2 = await adminDb.collection("warehouses").limit(500).get();
              warehouseCount = wh2.docs.filter((d: any) => (d.data() || {}).company_id === cid).length;
            } catch {
              warehouseCount = null;
            }
          }
          try {
            const pr = await adminDb.collection("products").where("company_id", "==", cid).get();
            productCount = pr.size;
          } catch (e: any) {
            details.push("products:" + (e?.message || "err"));
            try {
              const pr2 = await adminDb.collection("products").limit(500).get();
              productCount = pr2.docs.filter((d: any) => (d.data() || {}).company_id === cid).length;
            } catch {
              productCount = null;
            }
          }
          return {
            warehouseCount,
            productCount,
            detail: details.join(";") || "ok",
          };
        };
        const report = await buildInventoryMigrateDryRun(anStorageRouter, {
          companyId,
          countKemit,
        });
        res.json(report);
      } catch (err: any) {
        res.status(500).json({ ok: false, error: err?.message || String(err) });
      }
    });
```

```typescript
export async function fetchInventoryMigrateDryRun(companyId?: string): Promise<any> {

    return { ok: false, error: e?.message || String(e) };

}
```
