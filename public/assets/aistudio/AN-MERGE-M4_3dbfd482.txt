### 3.1 Create `src/lib/anApi.ts`
```typescript
/**
 * @file anApi.ts
 * @description MERGE-M4 — single namespace for A.N core HTTP surface.
 * All A.N endpoints MUST live under /api/an/*.
 * CustodyFlow endpoints stay under /api/auth, /api/admin, etc. — never duplicated under /api/an.
 */
export const AN_API_PREFIX = '/api/an' as const;

/** Catalog of mounted A.N routes (documentation + runtime discovery). */
export const AN_ROUTE_CATALOG = [
  {
    method: 'GET',
    path: '/api/an/storage-status',
    phase: 'M1',
    purpose: 'Probe StorageRouter / in-memory adapter health',
  },
  {
    method: 'POST',
    path: '/api/an/company-backend/:companyId',
    phase: 'M2',
    purpose: 'Validate and echo per-company BackendConfig binding',
  },
  {
    method: 'GET',
    path: '/api/an/auth-policy',
    phase: 'M3',
    purpose: 'Single login-path policy (firebase-session vs gated an-jwt)',
  },
  {
    method: 'GET',
    path: '/api/an/routes',
    phase: 'M4',
    purpose: 'List A.N API catalog (this registry)',
  },
] as const;

export type AnRouteEntry = (typeof AN_ROUTE_CATALOG)[number];

/** Build a path under the A.N namespace (rejects escaping the prefix). */
export function anUrl(subPath: string): string {
  const cleaned = subPath.startsWith('/') ? subPath : `/${subPath}`;
  if (cleaned.startsWith('/api/') && !cleaned.startsWith(AN_API_PREFIX)) {
    throw new Error(`A.N client must not call non-AN prefix: ${cleaned}`);
  }
  if (cleaned.startsWith(AN_API_PREFIX)) return cleaned;
  return `${AN_API_PREFIX}${cleaned}`;
}

/** CustodyFlow auth/admin prefixes that must remain outside /api/an. */
export const CF_RESERVED_PREFIXES = [
  '/api/auth',
  '/api/admin',
  '/api/migrations',
  '/api/seed',
] as const;

export function isAnNamespace(path: string): boolean {
  return path === AN_API_PREFIX || path.startsWith(`${AN_API_PREFIX}/`);
}

export function assertAnNamespace(path: string): void {
  if (!isAnNamespace(path)) {
    throw new Error(`Path is outside A.N namespace: ${path}`);
  }
}

```

### 3.2 Create `scripts/verifyMergeM4.ts`
```typescript
/**
 * verifyMergeM4.ts — A.N API namespace isolation
 */
import fs from 'fs';
import path from 'path';

let passed = 0;
let failed = 0;
function check(name: string, cond: boolean, detail = '') {
  if (cond) {
    console.log(`[PASS] ${name}`);
    passed++;
  } else {
    console.log(`[FAIL] ${name}${detail ? ' — ' + detail : ''}`);
    failed++;
  }
}

const root = process.cwd();
const apiPath = path.join(root, 'src/lib/anApi.ts');
const server = fs.readFileSync(path.join(root, 'server.ts'), 'utf8');
const bridge = fs.readFileSync(path.join(root, 'src/lib/anBridge.ts'), 'utf8');
const login = fs.readFileSync(path.join(root, 'src/pages/Login.tsx'), 'utf8');

check('S1 anApi.ts exists', fs.existsSync(apiPath));
const api = fs.readFileSync(apiPath, 'utf8');
check('S2 AN_API_PREFIX is /api/an', api.includes("'/api/an'") || api.includes('"/api/an"'));
check('S3 AN_ROUTE_CATALOG exported', api.includes('AN_ROUTE_CATALOG'));
check('S4 anUrl helper exported', api.includes('export function anUrl'));
check('S5 server mounts /api/an/routes', server.includes('/api/an/routes'));
check('S6 server keeps storage-status under /api/an', server.includes('/api/an/storage-status'));
check('S7 server keeps company-backend under /api/an', server.includes('/api/an/company-backend'));
check('S8 server keeps auth-policy under /api/an', server.includes('/api/an/auth-policy'));
check('S9 CF verify-credentials not under /api/an', server.includes('/api/auth/verify-credentials') && !server.includes('/api/an/auth/verify-credentials'));
check('S10 Login uses CF auth path', login.includes('/api/auth/verify-credentials'));
check('S11 anBridge only fetches /api/an for core probes', !bridge.includes("fetch('/api/auth"));
check('S12 no duplicate /api/an/storage-status definition twice-as-auth', (server.match(/\/api\/an\/storage-status/g) || []).length >= 1);
check('S13 catalog mentions M4 routes entry', api.includes("'/api/an/routes'") || api.includes('"/api/an/routes"'));
check('S14 verifyMergeM3 still present', fs.existsSync(path.join(root, 'scripts/verifyMergeM3.ts')));

console.log(`Passed: ${passed} | Failed: ${failed}`);
process.exit(failed > 0 ? 1 : 0);

```
