```typescript
/**
 * AN-INV-PILOT-ISSUE-1 — single ISSUE post for non-prod pilot.
 */
import type { StoragePort } from '../an-core/storage/StoragePort';
import type {
  StockDocument,
  StockDocumentLine,
  StockMovement,
  PostStockDocumentAtomicInput,
} from '../an-core/storage/StoragePort';
import { isPilotWritesEnabled } from './anInventoryPilotReceipt';

export interface PilotIssueInput {
  companyId: string;
  warehouseId: string;
  itemId: string;
  itemCode?: string;
  itemNameAr?: string;
  qty: number;
  unitCost?: number;
  idempotencyKey: string;
  actorId?: string;
}

export interface PilotIssueResult {
  ok: boolean;
  plane: 'an-inv-pilot-issue-1';
  writesEnabled: boolean;
  documentId?: string;
  error?: string;
  errorAr?: string;
}

export async function postPilotIssue(
  adapter: StoragePort,
  input: PilotIssueInput,
  env: NodeJS.ProcessEnv = process.env
): Promise<PilotIssueResult> {
  if (!isPilotWritesEnabled(env)) {
    return {
      ok: false,
      plane: 'an-inv-pilot-issue-1',
      writesEnabled: false,
      error: 'AN_INV_PILOT_WRITES not enabled',
      errorAr: 'كتابة الطيار غير مفعّلة',
    };
  }

  const companyId = input.companyId.trim();
  const warehouseId = input.warehouseId.trim();
  const itemId = input.itemId.trim();
  const qty = Number(input.qty);
  const unitCost = Number(input.unitCost ?? 0);
  const idempotencyKey = String(input.idempotencyKey || '').trim();
  const actorId = input.actorId || 'pilot';

  if (!companyId || !warehouseId || !itemId || !idempotencyKey) {
    return {
      ok: false,
      plane: 'an-inv-pilot-issue-1',
      writesEnabled: true,
      error: 'companyId, warehouseId, itemId, idempotencyKey required',
      errorAr: 'حقول إلزامية ناقصة',
    };
  }
  if (!(qty > 0)) {
    return {
      ok: false,
      plane: 'an-inv-pilot-issue-1',
      writesEnabled: true,
      error: 'qty must be > 0',
      errorAr: 'الكمية يجب أن تكون أكبر من صفر',
    };
  }

  const nowIso = new Date().toISOString();
  const docId = `pilot-iss-${idempotencyKey}`.slice(0, 120);
  const itemCode = input.itemCode || itemId;
  const itemNameAr = input.itemNameAr || itemCode;
  const totalCost = Math.round(qty * unitCost * 100) / 100;

  const line: StockDocumentLine = {
    id: `${docId}-L1`,
    tenantId: companyId,
    companyId,
    documentId: docId,
    lineNumber: 1,
    itemId,
    itemCode,
    itemNameAr,
    uom: 'unit',
    quantity: qty,
    unitCost,
    totalCost,
  };

  const movement: StockMovement = {
    id: `${docId}-M1`,
    tenantId: companyId,
    companyId,
    movementNumber: `${docId}-M1`,
    movementType: 'ISSUE',
    warehouseId,
    itemId,
    itemCode,
    itemNameAr,
    quantity: qty,
    unitCost,
    totalCost,
    balanceAfterQty: 0,
    sourceDocumentType: 'StockDocument',
    sourceDocumentId: docId,
    sourceDocumentNumber: docId,
    performedBy: actorId,
    performedByName: 'pilot-issue',
    timestamp: nowIso,
    idempotencyKey,
  };

  const doc = {
    id: docId,
    tenantId: companyId,
    companyId,
    documentNumber: docId,
    documentType: 'ISSUE' as const,
    warehouseId,
    status: 'POSTED' as const,
    totalQuantity: qty,
    totalCost,
    referenceType: 'PILOT_ISSUE',
    referenceId: idempotencyKey,
    notes: 'AN-INV-PILOT-ISSUE-1',
    lines: [line],
    postedAt: nowIso,
    postedBy: actorId,
    createdAt: nowIso,
    movements: [movement],
  };

  const payload: PostStockDocumentAtomicInput = {
    doc: doc as StockDocument,
    movements: [movement],
    allowNegative: false,
    useProvidedCost: true,
  };

  try {
    const res = await adapter.postStockDocumentAtomic(companyId, payload, idempotencyKey);
    if (!res.success) {
      return {
        ok: false,
        plane: 'an-inv-pilot-issue-1',
        writesEnabled: true,
        error: res.error || 'post_failed',
        errorAr: 'فشل ترحيل إذن الصرف (قد يكون الرصيد غير كافٍ)',
      };
    }
    return {
      ok: true,
      plane: 'an-inv-pilot-issue-1',
      writesEnabled: true,
      documentId: res.data?.id || docId,
    };
  } catch (e: any) {
    return {
      ok: false,
      plane: 'an-inv-pilot-issue-1',
      writesEnabled: true,
      error: (e?.message || String(e)).slice(0, 200),
      errorAr: 'استثناء أثناء الصرف',
    };
  }
}
```

If `isPilotWritesEnabled` is not exported from receipt module, export it there or duplicate the one-liner in this file.
